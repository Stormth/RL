# CarRacing-v3 (Double DQN) 实验依赖
# 建议使用 Python 3.10+
# 安装命令：pip install -r requirements.txt
#
# ⚠️ 版本兼容性（重要）：
#   box2d-py（CarRacing 的物理引擎）与旧版 matplotlib 均为 numpy 1.x ABI 编译，
#   若装成 numpy 2.x 会导致 matplotlib 报错 "_ARRAY_API not found"。
#   因此必须锁定 numpy<2；opencv-python 5.x 会强制升级 numpy 到 2.x，也要锁定 <5。
#
# 已验证可用的版本组合（Windows + conda）：
#   numpy 1.26.4 / opencv-python 4.10.0.84 / matplotlib 3.7.1
#   box2d-py 2.3.8 / gymnasium 1.3.0 / torch 2.5.1

numpy<2                    # 必须 <2（见上方说明）
opencv-python<5            # 必须 <5（避免被连带升级 numpy）
torch>=2.0                 # 深度学习框架（GPU 版请参照 PyTorch 官网安装）
gymnasium[box2d]>=1.0      # 强化学习环境（CarRacing-v3 依赖 Box2D 物理引擎）
matplotlib                 # 绘图
imageio                    # 生成 GIF（About_Env.ipynb 使用）
